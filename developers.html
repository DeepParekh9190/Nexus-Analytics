Track anything, anywhere

Our API is designed to be highly flexible. Track user actions, system events, or backend processes with a single function call.

  • ✓ Strongly typed payloads
  • ✓ Automatic retries on failure
  • ✓ Sub-millisecond latency
track.ts
import { NexusClient } from '@nexus/node';

const client = new NexusClient({
  apiKey: process.env.NEXUS_API_KEY,
  timeout: 5000
});

export async function handlePayment(userId, amount) {
  // Process the payment...
  
  await client.track({
    event: 'Payment Completed',
    userId: userId,
    properties: {
      revenue: amount,
      currency: 'USD'
    }
  });
}
